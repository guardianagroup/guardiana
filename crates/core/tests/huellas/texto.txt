def build(entries, base):
    n=len(entries); head=2+n*12+4
    data=b''; ents=b''
    for tag,typ,cnt,val in entries:
        if len(val)<=4:
            ents+=struct.pack('>HHI',tag,typ,cnt)+val.ljust(4,b'\0')
        else:
            off=base+head+len(data)
            ents+=struct.pack('>HHII',tag,typ,cnt,off)
            data+=val+(b'\0' if len(val)%2 else b'')
    return struct.pack('>H',n)+ents+b'\0\0\0\0'+data

def tiff(ifd0, exif, gps):
    tmp=build(ifd0+[(0x8769,4,1,b'\0\0\0\0'),(0x8825,4,1,b'\0\0\0\0')],8)
    exif_off=8+len(tmp)
    exif_b=build(exif,exif_off)
    gps_off=exif_off+len(exif_b)
    ifd=build(ifd0+[(0x8769,4,1,struct.pack('>I',exif_off)),(0x8825,4,1,struct.pack('>I',gps_off))],8)
    assert len(ifd)==len(tmp)
    return b'MM\0*\0\0\0\x08'+ifd+exif_b+build(gps,gps_off)

    objs=[b'<< /Type /Catalog /Pages 2 0 R /Metadata 5 0 R >>',
          b'<< /Type /Pages /Kids [3 0 R] /Count 1 >>',

    return o
